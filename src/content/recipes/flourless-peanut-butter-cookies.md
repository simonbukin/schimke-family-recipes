ingredients:
  - name: "Peanut butter"
    qty: "1"
    unit: "cup"
  - name: "Sugar"
    qty: "1/2"
    unit: "cup"
  - name: "Egg"
    qty: "1"
  - name: "Baking soda"
    qty: "1/2"
    unit: "teaspoon"
  - name: "Salt"
    qty: "1"
    unit: "pinch"
  - name: "Vanilla extract"
    qty: "1/2"
    unit: "teaspoon"
steps:
  - "Preheat oven to 350 degrees F."
  - "Combine the peanut butter, sugar, egg, baking soda, salt, and vanilla extract in a bowl and stir until well combined."
  - "Roll the dough into 1-inch balls and place on a parchment-lined baking sheet. Use a fork to press a crosshatch pattern into each cookie, flattening them slightly."
  - "Bake for 10-12 minutes until the edges are golden. Let cool on the baking sheet for a few minutes before transferring to a wire rack."
